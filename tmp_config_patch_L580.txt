    $p = str_replace(' ', '', strtolower($p));
    if ($p === '82x38_2box' || $p === '82x38-2box') {
        return [82.0, 38.0];
    }
    if (preg_match('/^(\d+(?:\.\d+)?)x(\d+(?:\.\d+)?)$/', $p, $m)) {
        return [(float) $m[1], (float) $m[2]];
    }
    return [(float) $fallbackW, (float) $fallbackH];
}

/** True when label preset is 82 mm × 38 mm sticker with two barcode boxes. */
function auragold_is_82x38_2box_sticker($preset): bool {
    $p = str_replace(' ', '', strtolower(trim((string) $preset)));
    return ($p === '82x38_2box' || $p === '82x38-2box');
}

/** Default geometry for 82×38 mm two-box sticker (mm). */
function auragold_82x38_2box_defaults(): array {
    return [
        'sticker_w'           => 82.0,
        'sticker_h'           => 38.0,
        'box_width_mm'        => 20.0,
        'box_height_mm'       => 25.0,
        'box1_left_mm'        => 4.0,
        'box1_top_mm'         => 7.0,
        'box2_left_mm'        => 58.0,
        'box2_top_mm'         => 2.0,
        'box_barcode_width_mm'  => 18.0,
        'box_barcode_height_mm' => 14.0,
    ];
}

/**
 * Resolved box layout for 82×38 two-box sticker from saved design_layout JSON.
 *
 * @return array{box_width_mm:float,box_height_mm:float,box1:array{left:float,top:float},box2:array{left:float,top:float},barcode_width_mm:float,barcode_height_mm:float}
 */
function auragold_82x38_2box_layout(array $snapshot = []): array {
    $def = auragold_82x38_2box_defaults();
    $boxW = (float) ($snapshot['box_width_mm'] ?? $snapshot['dual_quadrant_width_mm'] ?? $def['box_width_mm']);
    $boxH = (float) ($snapshot['box_height_mm'] ?? $snapshot['dual_quadrant_height_mm'] ?? $def['box_height_mm']);
    $boxW = max(8.0, min(40.0, $boxW));
    $boxH = max(8.0, min(40.0, $boxH));
    $b1L = (float) ($snapshot['box1_left_mm'] ?? $def['box1_left_mm']);
    $b1T = (float) ($snapshot['box1_top_mm'] ?? $def['box1_top_mm']);
    $b2L = (float) ($snapshot['box2_left_mm'] ?? $def['box2_left_mm']);
    $b2T = (float) ($snapshot['box2_top_mm'] ?? $def['box2_top_mm']);
    $stickerW = (float) ($def['sticker_w']);
    $stickerH = (float) ($def['sticker_h']);
    $b1L = max(0.0, min($stickerW - $boxW, $b1L));
    $b1T = max(0.0, min($stickerH - $boxH, $b1T));
    $b2L = max(0.0, min($stickerW - $boxW, $b2L));
    $b2T = max(0.0, min($stickerH - $boxH, $b2T));
    $barW = (float) ($snapshot['box_barcode_width_mm'] ?? $def['box_barcode_width_mm']);
    $barH = (float) ($snapshot['box_barcode_height_mm'] ?? $def['box_barcode_height_mm']);
    $barW = max(6.0, min($boxW, $barW));
    $barH = max(4.0, min($boxH - 4.0, $barH));
    return [
        'box_width_mm'       => round($boxW, 2),
        'box_height_mm'      => round($boxH, 2),
        'box1'               => ['left' => round($b1L, 2), 'top' => round($b1T, 2)],
        'box2'               => ['left' => round($b2L, 2), 'top' => round($b2T, 2)],
        'barcode_width_mm'   => round($barW, 2),
        'barcode_height_mm'  => round($barH, 2),
    ];
}

/**
 * One barcode box on an 82×38 mm sticker (print + screen preview).
 */
function render82x38BoxBarcode(string $code, int $boxNum, array $snapshot, bool $show_barcode_number): string {
    if ($code === '') {
        return '';
    }
    $layout = auragold_82x38_2box_layout($snapshot);
    $boxKey = ($boxNum === 2) ? 'box2' : 'box1';
    $pos = $layout[$boxKey];
    $boxW = $layout['box_width_mm'];
    $boxH = $layout['box_height_mm'];
    $graphicW = $layout['barcode_width_mm'];
    $graphicH = $layout['barcode_height_mm'];
    $style = 'left:' . $pos['left'] . 'mm;top:' . $pos['top'] . 'mm;width:' . $boxW . 'mm;height:' . $boxH . 'mm;';
    $graphicStyle = 'position:absolute;left:0;top:0;width:100%;height:calc(100% - 4mm);overflow:hidden;'
        . 'display:flex;align-items:center;justify-content:center;box-sizing:border-box;';
    $html = '<div class="barcode-box barcode-box--' . (int) $boxNum . '" style="' . htmlspecialchars($style, ENT_QUOTES, 'UTF-8') . '">';
    $html .= '<div class="barcode-82x38-graphic" style="' . htmlspecialchars($graphicStyle, ENT_QUOTES, 'UTF-8') . '"'
        . ' data-graphic-w-mm="' . $graphicW . '" data-graphic-h-mm="' . $graphicH . '">';
    $html .= '<svg class="barcode-svg barcode-svg--82x38" data-barcode="' . htmlspecialchars($code, ENT_QUOTES, 'UTF-8') . '"'
        . ' data-pocket-width-mm="' . $graphicW . '" data-pocket-height-mm="' . $graphicH . '"></svg>';
    $html .= '</div>';
    if ($show_barcode_number) {
        $html .= '<div class="barcode-box-text">' . htmlspecialchars($code, ENT_QUOTES, 'UTF-8') . '</div>';
    }
    $html .= '</div>';
    return $html;
}

/**
 * One 82×38 mm sticker page: box1 = left/first barcode, box2 = right/second barcode.
 *
 * @param array $print_item ['box1' => ['barcode','row'], 'box2' => ...] or legacy single
 */
function render82x38DoubleStickerLabel(array $print_item, array $settings, array $decoded_snapshot = []): string {
    $snapshot = is_array($decoded_snapshot) ? $decoded_snapshot : [];
    $show_number = isset($settings['show_barcode_number']) && (int) $settings['show_barcode_number'] === 1;
    $code1 = '';
    $code2 = '';
    if (isset($print_item['box1']) && is_array($print_item['box1'])) {
        $code1 = trim((string) ($print_item['box1']['barcode'] ?? ''));
    }
    if (isset($print_item['box2']) && is_array($print_item['box2'])) {
        $code2 = trim((string) ($print_item['box2']['barcode'] ?? ''));
    }
    if ($code1 === '' && $code2 === '') {
        $code1 = trim((string) ($print_item['barcode'] ?? ''));
    }
    if ($code1 === '' && $code2 === '') {
        return '';
    }
    $def = auragold_82x38_2box_defaults();
    $html = '<div class="sticker-page" data-sticker-w-mm="' . $def['sticker_w'] . '" data-sticker-h-mm="' . $def['sticker_h'] . '">';
    if ($code1 !== '') {
        $html .= render82x38BoxBarcode($code1, 1, $snapshot, $show_number);
    }
    if ($code2 !== '') {
        $html .= render82x38BoxBarcode($code2, 2, $snapshot, $show_number);
    }
    $html .= '</div>';
    return $html;
}
